    let clickTimer = null;
    const keysDB = JSON.parse(localStorage.getItem('eleven_keys')) || {};

    // DOM
    const body = document.body;
    const adminPanel = document.getElementById('adminPanel');
    const keyList = document.getElementById('keyList');
    const genKeyBtn = document.getElementById('genKeyBtn');
    const durationSelect = document.getElementById('durationSelect');
    const genResult = document.getElementById('genResult');
    const closeAdminBtn = document.getElementById('closeAdminBtn');

    const toggleGrid = document.getElementById('toggleGrid');
    const gameCards = document.querySelectorAll('.game-card');
    const tabs = document.querySelectorAll('.tab');
    const pages = {
      home: document.getElementById('home'),
      app: document.getElementById('app'),
      shop: document.getElementById('shop'),
      settings: document.getElementById('settings')
    };
    const keyInput = document.getElementById('keyInput');
    const keyBtn = document.getElementById('keyBtn');
    const keyStatus = document.getElementById('keyStatus');

    // ---- LƯU KEY ----
    function saveKeys() {
      localStorage.setItem('eleven_keys', JSON.stringify(keysDB));
    }

    function generateKey(length = 12) {
      const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
      let result = '';
      for (let i = 0; i < length; i++) {
        result += chars.charAt(Math.floor(Math.random() * chars.length));
      }
      return result;
    }

    function createKey(duration) {
      const key = generateKey();
      const now = new Date();
      let expiry = null;
      const durations = {
        '1ngay': 1,
        '1tuan': 7,
        '1thang': 30,
        '1nam': 365,
        'vinhvien': null
      };
      if (durations[duration] !== null) {
        const d = new Date(now);
        d.setDate(d.getDate() + durations[duration]);
        expiry = d.toISOString();
      }
      keysDB[key] = {
        created: now.toISOString(),
        expiry: expiry,
        status: 'active',
        duration: duration
      };
      saveKeys();
      renderKeyList();
      return key;
    }

    function deleteKey(key) {
      if (keysDB[key]) {
        delete keysDB[key];
        saveKeys();
        renderKeyList();
        return true;
      }
      return false;
    }

    function checkKey(key) {
      const data = keysDB[key];
      if (!data) return { valid: false, msg: 'Key không tồn tại!' };
      if (data.status === 'revoked') return { valid: false, msg: 'Key đã bị thu hồi!' };
      if (data.expiry) {
        const now = new Date();
        const exp = new Date(data.expiry);
        if (now > exp) return { valid: false, msg: 'Key đã hết hạn!' };
      }
      return { valid: true, msg: '✅ Key hợp lệ!' };
    }

    function renderKeyList() {
      const keys = Object.keys(keysDB);
      if (keys.length === 0) {
        keyList.innerHTML = '<div style="color:#666; text-align:center;">Chưa có key nào</div>';
        return;
      }
      let html = '';
      keys.forEach(k => {
        const data = keysDB[k];
        const expiry = data.expiry ? new Date(data.expiry).toLocaleDateString('vi') : 'Vĩnh viễn';
        const status = data.status === 'active' ? '🟢' : '🔴';
        html += `<div class="key-item">
          <span>${k} - ${expiry} ${status}</span>
          <span class="del" data-key="${k}">🗑️ Xoá</span>
        </div>`;
      });
      keyList.innerHTML = html;
      keyList.querySelectorAll('.del').forEach(el => {
        el.addEventListener('click', function() {
          const k = this.dataset.key;
          if (confirm(`Xoá key ${k}?`)) {
            deleteKey(k);
          }
        });
      });
    }

    // ---- MỞ ADMIN BẰNG CÁCH CLICK LOGO 5 LẦN ----
    document.getElementById('adminToggleLogo').addEventListener('click', function() {
      clickCount++;
      if (clickTimer) clearTimeout(clickTimer);
      clickTimer = setTimeout(() => {
        clickCount = 0;
      }, 1000);

      if (clickCount >= 5) {
        clickCount = 0;
        clearTimeout(clickTimer);
        // Chuyển trạng thái admin
        isAdmin = !isAdmin;
        if (isAdmin) {
          adminPanel.classList.add('show');
          renderKeyList();
          // Rung nhẹ logo
          this.style.animation = 'none';
          setTimeout(() => { this.style.animation = ''; }, 10);
        } else {
          adminPanel.classList.remove('show');
        }
      }
    });

    // ---- ĐÓNG ADMIN ----
    closeAdminBtn.addEventListener('click', function() {
      isAdmin = false;
      adminPanel.classList.remove('show');
    });

    // ---- TẠO KEY ----
    genKeyBtn.addEventListener('click', function() {

      const duration = durationSelect.value;
      const newKey = createKey(duration);
      genResult.textContent = `✅ Key mới: ${newKey}`;
      // Tự động copy key vào clipboard
      navigator.clipboard.writeText(newKey).then(() => {
        genResult.textContent = `✅ Key mới: ${newKey} (đã copy)`;
      }).catch(() => {});
      setTimeout(() => { genResult.textContent = ''; }, 5000);
    });

    // ---- TOGGLE ----
    TOGGLES.forEach(name => {
      const card = document.createElement('div');
      card.className = 'toggle-card';
      const label = document.createElement('div');
      label.className = 'label';
      label.textContent = name;
      const btn = document.createElement('button');
      btn.className = 'toggle-btn';
      btn.setAttribute('aria-label', `Toggle ${name}`);
      const dot = document.createElement('span');
      dot.className = 'dot';
      btn.appendChild(dot);
      btn.dataset.state = 'off';
      btn.addEventListener('click', function(e) {
        e.stopPropagation();
        if (!isUnlocked) {
          keyStatus.textContent = '⚠️ Vui lòng nhập key kích hoạt trước!';
          keyStatus.style.color = '#ff3333';
          return;
        }
        const isOn = this.dataset.state === 'on';
        this.dataset.state = isOn ? 'off' : 'on';
        this.classList.toggle('on', !isOn);
        try {
          const ctx = new (window.AudioContext || window.webkitAudioContext)();
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.frequency.value = 880;
          osc.type = 'sine';
          gain.gain.value = 0.06;
          osc.start();
          setTimeout(() => { osc.stop(); ctx.close(); }, 100);
        } catch(e) {}
      });
      card.appendChild(label);
      card.appendChild(btn);
      toggleGrid.appendChild(card);
    });

    // ---- GAME ----
    gameCards.forEach(card => {
      card.addEventListener('click', function(e) {
        if (!isUnlocked) {
          keyStatus.textContent = '⚠️ Cần key kích hoạt để chọn game!';
          keyStatus.style.color = '#ff3333';
          return;
        }
        gameCards.forEach(c => c.classList.remove('selected'));
        this.classList.add('selected');
        const game = this.dataset.game;
        let url = game === 'freefire' ? 'freefire://' : 'freefiremax://';
        const iframe = document.createElement('iframe');
        iframe.style.display = 'none';
        iframe.src = url;
        document.body.appendChild(iframe);
        setTimeout(() => {
          if (document.hidden !== undefined) {
            window.location.href = 'https://apps.apple.com/app/free-fire/id1300146617';
          }
          iframe.remove();
        }, 800);
      });
    });

    // ---- TAB ----
    tabs.forEach(tab => {
      tab.addEventListener('click', function() {
        const pageId = this.dataset.page;
        if (!isUnlocked && pageId !== 'home') {
          keyStatus.textContent = '⚠️ Cần key kích hoạt để truy cập!';
          keyStatus.style.color = '#ff3333';
          return;
        }
        tabs.forEach(t => t.classList.remove('active'));
        this.classList.add('active');
        Object.keys(pages).forEach(key => {
          pages[key].classList.toggle('active', key === pageId);
        });
        document.getElementById('contentArea').scrollIntoView({ behavior: 'smooth', block: 'start' });
      });
    });

    // ---- USER KEY ----
    keyBtn.addEventListener('click', function() {
      const input = keyInput.value.trim().toUpperCase();
      if (!input) {
        keyStatus.textContent = '⚠️ Vui lòng nhập key!';
        keyStatus.style.color = '#ff3333';
        return;
      }
      const result = checkKey(input);
      keyStatus.textContent = result.msg;
      keyStatus.style.color = result.valid ? '#66ff66' : '#ff3333';
      if (result.valid) {
        isUnlocked = true;
        body.classList.remove('locked');
        keyInput.value = '';
        document.querySelectorAll('.tab').forEach(t => t.style.pointerEvents = 'auto');
        document.querySelectorAll('.toggle-card, .game-card, .settings-item, .price-item').forEach(el => {
          el.style.opacity = '1';
          el.style.pointerEvents = 'auto';
        });
        document.getElementById('keySection').style.display = 'none';
      } else {
        keyInput.value = '';
        setTimeout(() => { keyStatus.textContent = ''; }, 2500);
      }
    });

    keyInput.addEventListener('keydown', function(e) {
      if (e.key === 'Enter') {
        e.preventDefault();
        keyBtn.click();
      }
    });

    // Ngăn nhấp nháy khi copy/dán
    keyInput.addEventListener('focus', function() {
      this.setSelectionRange(0, this.value.length);
    });

    // ---- KHỞI TẠO ----
    body.classList.add('locked');
    if (!keysDB['ELEVEN2026']) {
      keysDB['ELEVEN2026'] = {
        created: new Date().toISOString(),
        expiry: null,
        status: 'active',
        duration: 'vinhvien'
      };
      saveKeys();
    }
    renderKeyList();

    // ---- CHẶN KÉO HEADER ----
    const header = document.getElementById('header');
    header.addEventListener('touchmove', function(e) {
      e.preventDefault();
    }, { passive: false });

    document.addEventListener('touchmove', function(e) {
      if (e.target.closest('.header')) {
        e.preventDefault();
      }
    }, { passive: false });

    console.log('💀 Devils Will Rise – Admin mở bằng logo (5 lần click)');
  })();
</script>
